      function drawTimelinesForTestFirst(){
        var container = document.getElementById('timeline-testFirst');
        var chart = new google.visualization.Timeline(container);
        var dataTable = new google.visualization.DataTable();
        dataTable.addColumn({ type: 'string', id: 'File' });
        dataTable.addColumn({ type: 'string', id: 'Action' });
        dataTable.addColumn({ type: 'string', id: 'style', role:'style' });
        dataTable.addColumn({ type: 'number', id: 'Start' });
        dataTable.addColumn({ type: 'number', id: 'End' });
        var data = ##%%$$TESTFIRSTDATA$$%%##;
        var durations = ##%%$$DURATION$$%%##;
        var oneToZero = ##%%$$ONETOZERO$$%%##;
        var twoToZero = ##%%$$TWOTOZERO$$%%##;
        var calcTimeZone = 15 * 60 * 60 * 1000;
        var filteredData =
          data
            .map((row) => {
              let color;
              const start = row[2];
              const isWithinDuration = durations.some(([dStart, dEnd]) => {
                return start >= dStart && start <= dEnd;
              });
              const isInOneToZero = oneToZero.some(([dStart, dEnd]) => {
                return start >= dStart && start <= dEnd;
              });
              const isInTwoToZero = twoToZero.some(([dStart, dEnd]) => {
                return start >= dStart && start <= dEnd;
              });
              if(isWithinDuration){
                color = "#4169e1";
              }else if(isInOneToZero){
                color = "#ffa500";
              }else if(isInTwoToZero){
                color = "#ffc0cb";
              }else{
                color = "#ff0000";
              }
              return [row[0], row[1], color , row[2] +  calcTimeZone,row[3] + calcTimeZone];
            });
        dataTable.addRows(
          filteredData
        );
        var options = {
            timeline: {
              showBarLabels: false
            },
            width: 850,
            height: 350,
            hAxis: {
              minValue:new Date(15 * 60 * 60 * 1000)
            },
          };
        chart.draw(dataTable,options);
      }
